#include <stdio.h>
#include <math.h>

int main() {
    float shirina, visota, zapas-diametru;

    printf("Доброго дня. Програма допоможе порахувати розмір отвору, "
           "який необхідний для вільного проходження в ньому "
           "прямокутної деталі.\n");

    printf("Введіть ширину: ");
    scanf("%f", &shirina);

    printf("Введіть висоту: ");
    scanf("%f", &visota);

    printf("Введіть запас діаметру: ");
    scanf("%f", &zapas_diametru);

    float result = sqrt(pow(shirina, 2) + visota * visota);
    float nareshti = result * (1 + zapas-diametru / 100);

    printf("Діаметри: %f\n", result);
    printf("Надлишок: %f\n", nareshti);

    return 0;
}

\end{lstlisting}

\textbf{Висновок:} При написанні коду використовував модуль \texttt{<math.h>}.
